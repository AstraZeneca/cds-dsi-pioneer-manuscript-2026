**Goal:** Restore interval censoring for the 0→1 (progression) multistate transition by marginalizing the likelihood over all possible true progression weeks between the last clean assessment and the detection visit.

**Architecture:** Add `interval_censored` as a Stan data input; derive `ms_ic_gap_01` in `transformed data`; extend `multistate_lpmf` with an IC marginalization loop using cumulative-sum precomputation and an incremental sojourn update trick. Passing all-zeros for `ms_ic_gap_01` recovers the current no-IC behavior exactly, enabling a clean validation path.

```r
test_that("IC marginalization matches hand-computed expected value", {
  # State 2 patient: T_c=4, T_d=6 (gap=2), T_death=10
  # time_12 = T_death - T_d = 4
  # Constant log-conditional-survival = -0.1 for all transitions
  c_val <- -0.1
  c_h   <- log1p(-exp(c_val))  # log(1 - exp(-0.1)) = log(hazard prob)

  # base_surv: survived all 3 risks weeks 1..T_c=4
  base_surv <- 3 * 4 * c_val  # = -1.2

  # For each candidate s in {5, 6} (k=1,2 with T_c=4, T_d=6):
  # alpha(s): survived 0->1 from T_c+1 to s-1, event at s
  # beta(s):  survived 02+03 from T_c+1 to s-1
  # gamma(s): sojourn event at tau_s = time_12 + (T_d - s)

  # k=1, s=5: alpha = c_h, beta = 0, tau_s = 4+1 = 5
  # sojourn event at 5: sum(lcs[1:4]) + log1m_exp(lcs[5]) = 4*c_val + c_h
  term1 <- c_h + 0 + (4 * c_val + c_h)

  # k=2, s=6: alpha = c_val + c_h, beta = 2*c_val, tau_s = 4+0 = 4
  # sojourn event at 4: sum(lcs[1:3]) + log1m_exp(lcs[4]) = 3*c_val + c_h
  term2 <- (c_val + c_h) + (2 * c_val) + (3 * c_val + c_h)

  expected_ll <- base_surv + matrixStats::logSumExp(c(term1, term2))

  data <- list(
    n_patients    = 1L,
    max_t         = 30L,
    max_sojourn_t = 15L,
    final_state   = 2L,
    time_01       = 6L,
    time_02       = 0L,

    "tests/testthat/stan/test_multistate_ic.stan", data

})
```

### Mathematical recap

For a progressed patient with last-clean week $T_c$, detection week $T_d$, gap $= T_d - T_c$:

```
log L = base_surv(1..T_c)
      + log_sum_exp over s = T_c+1..T_d of:
          alpha(s) = surv_01(T_c+1..s-1) + log_h_01(s)
          beta(s)  = surv_02+03(T_c+1..s-1)
          gamma(s) = sojourn_ll(tau_s)   where tau_s = T_end - s
```

The sojourn is computed **incrementally**: seed at $s = T_d$ (tau = `time_12`), then prepend one week per step.

```stan
else if (final_state[i] == 1) {
  int T_d  = time_01[i];
  int gap  = ms_ic_gap_01[i];
  int T_c  = T_d - gap;

  // Base survival through last clean assessment (certain for all candidates)
  if (enable_01 && T_c > 0) patient_ll += sum(log_cond_surv_01[i, 1:T_c]);
  if (enable_02 && T_c > 0) patient_ll += sum(log_cond_surv_02[i, 1:T_c]);
  if (enable_03 && T_c > 0) patient_ll += sum(log_cond_surv_03[i, 1:T_c]);

  if (gap == 0 || prog_deterministic[i]) {
    // ── No-IC path (gap=0 or deterministic PD) ──────────────────────────
    // Survival from T_c+1..T_d-1 and event at T_d
    if (enable_01 && T_d > T_c + 1)
      patient_ll += sum(log_cond_surv_01[i, (T_c + 1):(T_d - 1)]);
    if (enable_02 && T_d > T_c + 1)
      patient_ll += sum(log_cond_surv_02[i, (T_c + 1):(T_d - 1)]);
    if (enable_03 && T_d > T_c + 1)
      patient_ll += sum(log_cond_surv_03[i, (T_c + 1):(T_d - 1)]);
    if (enable_01 && !prog_deterministic[i] && T_d > 0)
      patient_ll += log1m_exp(log_cond_surv_01[i, T_d]);
    // Post-progression sojourn (censored) — inline from existing code
    if (enable_12 && time_12[i] > 0) {
      if (ms_time_scale_12 == 1) {
        patient_ll += sum(log_cond_surv_12_s[i, 1:time_12[i]]);
      } else if (ms_time_scale_12 == 0) {
        int t_start = T_d + 1;
        int t_end   = T_d + time_12[i];
        if (t_end <= cols(log_cond_surv_12_t))
          patient_ll += sum(log_cond_surv_12_t[i, t_start:t_end]);
      } else if (ms_time_scale_12 == 2) {
        patient_ll += sum(log_cond_surv_12_s[i, 1:time_12[i]]);
        int t_start = T_d + 1;
        int t_end   = T_d + time_12[i];
        if (t_end <= cols(log_cond_surv_12_t))
          patient_ll += sum(log_cond_surv_12_t[i, t_start:t_end]);
      }
    }

  } else {
    // ── IC path: marginalize over true progression week s ───────────────
    // Precompute cumulative sums over IC interval [T_c+1, T_d]
    vector[gap] cum_01 = cumulative_sum(
      to_vector(log_cond_surv_01[i, (T_c + 1):T_d]));
    vector[gap] cum_02 = enable_02 ? cumulative_sum(
      to_vector(log_cond_surv_02[i, (T_c + 1):T_d])) : rep_vector(0.0, gap);
    vector[gap] cum_03 = enable_03 ? cumulative_sum(
      to_vector(log_cond_surv_03[i, (T_c + 1):T_d])) : rep_vector(0.0, gap);

    vector[gap] terms;

    // Seed sojourn at k=gap (s=T_d, tau=time_12) — censored state 1
    real sojourn_ll = enable_12 && time_12[i] > 0
      ? sum(log_cond_surv_12_s[i, 1:time_12[i]])
      : 0.0;
    // Note: for Markov/Extended, seed differently (see Task 8)

    for (kk in 0:(gap - 1)) {
      int k = gap - kk;       // k: gap, gap-1, ..., 1
      int s = T_c + k;
      int tau_s = time_12[i] + (T_d - s);  // = time_12 + kk

      // Update incremental sojourn (append one survival week, censored case)
      if (kk > 0 && enable_12)
        sojourn_ll += log_cond_surv_12_s[i, tau_s];

      real alpha = (k > 1 ? cum_01[k - 1] : 0.0)
                 + log1m_exp(log_cond_surv_01[i, s]);
      real beta  = (k > 1 ? cum_02[k - 1] + cum_03[k - 1] : 0.0);

      terms[k] = alpha + beta + sojourn_ll;
    }

    patient_ll += log_sum_exp(terms);
  }
}
```

```stan
    if (enable_12 && time_12[i] > 0) {
      if (ms_time_scale_12 == 1) {
        if (time_12[i] > 1)
          patient_ll += sum(log_cond_surv_12_s[i, 1:(time_12[i] - 1)]);
        patient_ll += log1m_exp(log_cond_surv_12_s[i, time_12[i]]);
      } else if (ms_time_scale_12 == 0) {
        int t_start = T_d + 1;
        int t_death = T_d + time_12[i];
        if (t_death - 1 >= t_start && t_death <= cols(log_cond_surv_12_t))
          patient_ll += sum(log_cond_surv_12_t[i, t_start:(t_death - 1)]);
        if (t_death <= cols(log_cond_surv_12_t))
          patient_ll += log1m_exp(log_cond_surv_12_t[i, t_death]);
      } else if (ms_time_scale_12 == 2) {
        if (time_12[i] > 1)
          patient_ll += sum(log_cond_surv_12_s[i, 1:(time_12[i] - 1)]);
        int t_start = T_d + 1;
        int t_death = T_d + time_12[i];
        if (t_death - 1 >= t_start && t_death <= cols(log_cond_surv_12_t))
          patient_ll += sum(log_cond_surv_12_t[i, t_start:(t_death - 1)]);
        real combined_lcs = log_cond_surv_12_s[i, time_12[i]];
        if (t_death <= cols(log_cond_surv_12_t))
          combined_lcs += log_cond_surv_12_t[i, t_death];
        patient_ll += log1m_exp(combined_lcs);
      }
    }

  } else {
    // ── IC path ──────────────────────────────────────────────────────
    vector[gap] cum_01 = cumulative_sum(
      to_vector(log_cond_surv_01[i, (T_c + 1):T_d]));
    vector[gap] cum_02 = enable_02 ? cumulative_sum(
      to_vector(log_cond_surv_02[i, (T_c + 1):T_d])) : rep_vector(0.0, gap);
    vector[gap] cum_03 = enable_03 ? cumulative_sum(
      to_vector(log_cond_surv_03[i, (T_c + 1):T_d])) : rep_vector(0.0, gap);

    vector[gap] terms;

    // Seed sojourn at k=gap (s=T_d, tau=time_12) — event case
    real sojourn_ll = 0;
    if (enable_12 && time_12[i] > 0) {
      if (time_12[i] > 1)
        sojourn_ll += sum(log_cond_surv_12_s[i, 1:(time_12[i] - 1)]);
      sojourn_ll += log1m_exp(log_cond_surv_12_s[i, time_12[i]]);
    }

    for (kk in 0:(gap - 1)) {
      int k = gap - kk;
      int s = T_c + k;
      int tau_s = time_12[i] + (T_d - s);  // = time_12 + kk

      // Update incremental sojourn (event moves one week forward):
      // old event at tau_s-1 becomes survival; new event at tau_s
      if (kk > 0 && enable_12) {
        sojourn_ll -= log1m_exp(log_cond_surv_12_s[i, tau_s - 1]);
        sojourn_ll += log_cond_surv_12_s[i, tau_s - 1];
        sojourn_ll += log1m_exp(log_cond_surv_12_s[i, tau_s]);
      }

      real alpha = (k > 1 ? cum_01[k - 1] : 0.0)
                 + log1m_exp(log_cond_surv_01[i, s]);
      real beta  = (k > 1 ? cum_02[k - 1] + cum_03[k - 1] : 0.0);

      terms[k] = alpha + beta + sojourn_ll;
    }

    patient_ll += log_sum_exp(terms);
  }
}
```

> **Note on Markov / Extended time scales (`ms_time_scale_12 = 0 or 2`):** The above code uses `log_cond_surv_12_s` (semi-Markov sojourn GP). For `ms_time_scale_12 = 0` (Markov) and `ms_time_scale_12 = 2` (Extended), the sojourn likelihood also involves `log_cond_surv_12_t` indexed by calendar week. The IC loop shifts the clock-forward start as well as the sojourn length. Since SCLC-01 currently uses semi-Markov (`ms_time_scale_12 = 1`), implement Markov and Extended cases as `gap == 0` (no IC) for now with a TODO comment. Tracking issue for full Markov/Extended IC support can be opened separately.

### Why visit-conditioning?

Dropout (0→3) can only be detected at an assessment visit. Between visits the patient is necessarily still on trial — it is impossible to observe dropout at a non-visit week. The 0→3 hazard should therefore only accumulate at assessment visit weeks, not at every calendar week. Admin-censored patients (state 0) are censored by the data cutoff, not by dropout, and contribute **nothing** to the 0→3 hazard.

`t_patient_visits` (flat array, all patients) and `patient_visit_pos` (position array, length `n_patients + 1`) already exist as program-level variables in `_base_transformed_data.stan`. They are passed as new function arguments — no new data declarations are needed.
